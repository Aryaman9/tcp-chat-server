cmake_minimum_required(VERSION 3.20)
project(tcp_chat_server LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This project requires Linux. On Windows, build inside WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(ENABLE_SANITIZERS AND ENABLE_TSAN)
  message(FATAL_ERROR "Use separate builds for ASan/UBSan and TSan")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(ENABLE_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_executable(chat_server src/server.cpp)
target_include_directories(chat_server PRIVATE include)
target_link_libraries(chat_server PRIVATE Threads::Threads)
add_executable(chat_client src/client.cpp)
target_include_directories(chat_client PRIVATE include)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_include_directories(core_tests PRIVATE include)
  target_link_libraries(core_tests PRIVATE Threads::Threads)
  add_test(NAME core COMMAND core_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_network.py $<TARGET_FILE:chat_server> $<TARGET_FILE:chat_client>)
  set_tests_properties(core PROPERTIES TIMEOUT 15)
  set_tests_properties(network PROPERTIES TIMEOUT 90)
endif()
